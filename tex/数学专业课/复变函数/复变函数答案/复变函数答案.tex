	\textbf{16.}\quad 证明:\\[4pt]
	\textbf{一、$\partial E\cup E\subseteq E'$}\\[2pt]
	$1^\circ$ $E\subseteq E'$:\\
	设 $z_0\in E$.\\
	$\because E$ 为开集,\\
	$\therefore\exists r>0$ 使 $B(z_0,r)\subseteq E$.\\
	$\forall\varepsilon>0$, 取 $w=z_0+\frac{1}{2}\min(r,\varepsilon)$,\\
	则 $0<|w-z_0|<\min(r,\varepsilon)$,\\
	$\therefore w\in E$ 且 $w\in B(z_0,\varepsilon)\setminus\{z_0\}$,\\
	$\therefore z_0\in E'$, 即 $E\subseteq E'$.\\[4pt]
	$2^\circ$ $\partial E\subseteq E'$:\\
	设 $z_0\in\partial E$.\\
	若 $z_0\in E$, 由 $1^\circ$ 知 $z_0\in E'$;\\
	若 $z_0\notin E$,\\
	$\because z_0\in\partial E$,\\
	$\therefore\forall\varepsilon>0$, $B(z_0,\varepsilon)\cap E\neq\varnothing$,\\
	且其中各点均异于 $z_0$,\\
	$\therefore\big(B(z_0,\varepsilon)\setminus\{z_0\}\big)\cap E\neq\varnothing$,\\
	$\therefore z_0\in E'$.\\
	综上, $\partial E\cup E\subseteq E'$.\\[4pt]
	\textbf{二、$E'\subseteq\partial E\cup E$}\\[2pt]
	设 $z_0\in E'$.\\
	若 $z_0\in E$, 结论已成立;\\
	若 $z_0\notin E$,\\
	$\because z_0\in E'$,\\
	$\therefore\forall\varepsilon>0$, $\big(B(z_0,\varepsilon)\setminus\{z_0\}\big)\cap E\neq\varnothing$,\\
	$\therefore B(z_0,\varepsilon)\cap E\neq\varnothing$;\\
	又 $\because z_0\notin E$,\\
	$\therefore z_0\in B(z_0,\varepsilon)\cap E^c$,\\
	$\therefore B(z_0,\varepsilon)\cap E^c\neq\varnothing$,\\
	$\therefore z_0\in\partial E$.\\
	综上, $E'\subseteq\partial E\cup E$.\\[4pt]
	综合一、二, $E'=\partial E\cup E$. $\blacksquare$

	\textbf{4.}\quad 解: 设 $z=x+iy$, $f(z)=u(x,y)+iv(x,y)$, Cauchy-Riemann 条件为
	\[
	\frac{\partial u}{\partial x}=\frac{\partial v}{\partial y},\qquad
	\frac{\partial u}{\partial y}=-\frac{\partial v}{\partial x},
	\quad\text{即}\quad u_x=v_y,\ u_y=-v_x.
	\]
	
	\textbf{(1)} $\sin z$:\\
	$\because \sin z=\sin x\cosh y+i\cos x\sinh y$,\\
	$\therefore u=\sin x\cosh y$,\quad $v=\cos x\sinh y$.\\
	$\because u_x=\cos x\cosh y=v_y$,\\
	\quad $u_y=\sin x\sinh y=-v_x$,\\
	$\therefore$ 满足 C-R 条件.\\[4pt]
	
	\textbf{(2)} $\cos z$:\\
	$\because \cos z=\cos x\cosh y-i\sin x\sinh y$,\\
	$\therefore u=\cos x\cosh y$,\quad $v=-\sin x\sinh y$.\\
	$\because u_x=-\sin x\cosh y=v_y$,\\
	\quad $u_y=\cos x\sinh y=-v_x$,\\
	$\therefore$ 满足 C-R 条件.\\[4pt]
	
	\textbf{(3)} $z^3$:\\
	$\because z^3=(x^3-3xy^2)+i(3x^2y-y^3)$,\\
	$\therefore u=x^3-3xy^2$,\quad $v=3x^2y-y^3$.\\
	$\because u_x=3x^2-3y^2=v_y$,\\
	\quad $u_y=-6xy=-v_x$,\\
	$\therefore$ 满足 C-R 条件.\\[4pt]
	
	\textbf{(4)} $\mathrm{e}^z$:\\
	$\because \mathrm{e}^z=\mathrm{e}^x\cos y+i\,\mathrm{e}^x\sin y$,\\
	$\therefore u=\mathrm{e}^x\cos y$,\quad $v=\mathrm{e}^x\sin y$.\\
	$\because u_x=\mathrm{e}^x\cos y=v_y$,\\
	\quad $u_y=-\mathrm{e}^x\sin y=-v_x$,\\
	$\therefore$ 满足 C-R 条件.\\[4pt]
	
	综上, 四个函数的实部与虚部在 $\mathbb{C}$ 上处处满足 C-R 条件.

	\textbf{5.}\quad 证明:\\[4pt]
	\textbf{引理（常值判别）}: 若实值函数 $u$ 在区域 $\Omega$ 内恒有 $u_x=u_y=0$, 则 $u$ 在 $\Omega$ 内为常数.\\
	$\because\Omega$ 为连通开集,\\
	$\therefore\forall z_0,z_1\in\Omega$, 可用完全含于 $\Omega$ 的折线联结.\\
	每段折线上以 $t$ 为参数, 则 $\dfrac{\mathrm{d}u}{\mathrm{d}t}=u_x\,x'(t)+u_y\,y'(t)=0$,\\
	$\therefore u(z_0)=u(z_1)$, 即 $u$ 为常数.\\[4pt]
	
	\textbf{(1)} 设 $f=u+\mathrm{i}v$.\\
	$\because f'=u_x+\mathrm{i}v_x=0$,\\
	$\therefore u_x=v_x=0$;\\
	又 $\because f$ 解析, $\therefore$ 满足 C-R 条件: $u_y=-v_x$, $v_y=u_x$,\\
	$\therefore u_y=v_y=0$.\\
	$\therefore u,v$ 的一阶偏导在 $\Omega$ 内恒为零,\\
	由引理, $u,v$ 均为常数, $\therefore f$ 在 $\Omega$ 内为常数.\\[4pt]
	
	\textbf{(2)} 设 $f=u+\mathrm{i}v$, 则 $|f|^2=u^2+v^2=c^2$ ($c\geq0$ 为常数).\\
	若 $c=0$, 则 $f\equiv0$, 为常数.\\
	下设 $c>0$. 两边分别对 $x,y$ 求偏导:\\
	$\because uu_x+vv_x=0$,\\
	\quad $uu_y+vv_y=0$,\\
	将 C-R 条件 $u_y=-v_x$, $v_y=u_x$ 代入第二式, 得 $vu_x-uv_x=0$.\\
	$\therefore$
	\[
	\begin{pmatrix}u&v\\ v&-u\end{pmatrix}
	\begin{pmatrix}u_x\\ v_x\end{pmatrix}=
	\begin{pmatrix}0\\0\end{pmatrix},
	\]
	$\because$ 系数行列式 $=-(u^2+v^2)=-c^2\neq0$,\\
	$\therefore u_x=v_x=0$, 进而 $u_y=v_y=0$.\\
	由引理, $u,v$ 均为常数, $\therefore f$ 在 $\Omega$ 内为常数. $\blacksquare$
	
\end{minipage}
