\section{第一讲\quad 偏微分方程的基本概念}
\subsection{PDE 的定义与记号}
\begin{knowledge}{偏微分方程（PDE）}
	
	包含一个未知函数、两个或更多自变量以及该未知函数偏导数的方程，称为偏微分方程.
	
	记号约定：
	\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
		
		\item $U$ 为 $\mathbb{R}^n$ 中的开集，$U \subseteq \mathbb{R}^n$.
		\item 未知函数 $u: U \to \mathbb{R}$，$u(x) = u(x_1, x_2, \dots, x_n)$，$x = (x_1,\dots,x_n) \in U$.
		\item 偏导数：$u_{x_i} = \dfrac{\partial u}{\partial x_i}$，$u_{x_i x_j} = \dfrac{\partial^2 u}{\partial x_j \partial x_i}$.
	\end{enumerate}
\end{knowledge}
\begin{knowledge}{多重指标记号}
	
	设 $\alpha = (\alpha_1, \dots, \alpha_n)$ 为 $n$ 维非负整数组（称为多重指标），其阶数定义为
	\[
	|\alpha| = \alpha_1 + \dots + \alpha_n.
	\]
	相应的高阶偏导数记为
	\[
	D^\alpha u = \frac{\partial^{|\alpha|} u}{\partial^{\alpha_1} x_1 \, \partial^{\alpha_2} x_2 \cdots \partial^{\alpha_n} x_n} = \frac{\partial^{|\alpha|} u}{\partial x_1^{\alpha_1} \cdots \partial x_n^{\alpha_n}}.
	\]
	并把所有 $k$ 阶偏导数构成的集合记为
	\[
	D^k u = \{ D^\alpha u : |\alpha| = k \}.
	\]
	特别地，一阶偏导数组成的向量称为梯度：
	\[
	Du = (u_{x_1}, \dots, u_{x_n}) = \nabla u.
	\]
\end{knowledge}
\subsection{线性偏微分方程}
\begin{knowledge}{线性与齐次}
	
	形如
	\[
	\sum_{|\alpha| \leq k} a_\alpha(x) D^\alpha u(x) = f(x)
	\]
	的方程称为 $k$ 阶线性偏微分方程，其中 $a_\alpha$，$f$ 为给定函数.
	
	"线性"指未知函数 $u$ 及其各阶偏导数均以一次幂线性出现，方程中不出现 $\|Du\|^2$，$\sin u$ 等非线性项.
	
	若 $f \equiv 0$，称该 PDE 为齐次的；否则称为非齐次的.
\end{knowledge}

\section{第二讲\quad 传输方程}
\subsection{齐次方程与特征线法}
\begin{knowledge}{传输方程}
	
	设 $\vec{b} = (b_1,\dots,b_n) \in \mathbb{R}^n$ 为常向量，传输方程为
	\[
	u_t + \vec{b} \cdot Du = 0, \quad (x,t) \in \mathbb{R}^n \times (0, +\infty).
	\]
	改写为内积形式
	\[
	(1, \vec{b}) \cdot (u_t, Du) = 0,
	\]
	即 $u$ 沿空间-时间方向 $(1, \vec{b})$ 的方向导数为零. 特征线法的思想是：选取适当曲线（特征线），把 PDE 化为沿该曲线的常微分方程.
\end{knowledge}
\category{特征线法}
\begin{question}{沿特征线求导}
	
	取特征线 $x(t) = \vec{b} t + x_0$（即 $x = bt + x_0$），验证 $u$ 沿特征线为常数.
\end{question}

\begin{minipage}[t]{0.7\textwidth}
	
	沿特征线对 $t$ 求全导数：
	\[
	\frac{d}{dt} u(x(t), t) = \sum_{i=1}^n u_{x_i}(x(t), t) \cdot \frac{d x_i(t)}{dt} + u_t(x(t), t)
	\]
	\[
	= \sum_{i=1}^n b_i u_{x_i}(x(t), t) + u_t(x(t), t) = \vec{b} \cdot Du + u_t = 0.
	\]
	$\therefore u(x(t), t)$ 沿特征线为常数.
	
	$\therefore u(x, t) = u(x_0, 0)$，其中由 $x = \vec{b} t + x_0$ 得 $x_0 = x - \vec{b} t$，即
	\[
	u(x, t) = u(x - \vec{b} t, 0), \quad \forall x \in \mathbb{R}^n,\ t > 0.
	\]
\end{minipage}

\subsection{初值问题与行波解}
\begin{knowledge}{初值问题}
	
	考虑初值问题
	\[
	\begin{cases}
		u_t + \vec{b} \cdot Du = 0 & \mathbb{R}^n \times (0,+\infty) \\
		u = g & \mathbb{R}^n \times \{t=0\}
	\end{cases}
	\]
	即 $t=0$ 时 $u(x,0) = g(x)$. 由特征线结论立得
	\[
	u(x,t) = g(x - \vec{b} t).
	\]
	物理意义：初始波形 $g$ 以速度 $\vec{b}$ 整体平移，故称行波解.
\end{knowledge}
\begin{question}{验证}
	
	$u(x,t) = g(x-\vec{b}t)$ 确为方程的解.
\end{question}

由链式法则 $u_t = -\vec{b} \cdot Dg(x-\vec{b}t)$，$Du = Dg(x-\vec{b}t)$，
\[
u_t + \vec{b} \cdot Du = -\vec{b} \cdot Dg(x-\vec{b}t) + \vec{b} \cdot Dg(x-\vec{b}t) = 0.
\]
\subsection{变系数情形}
\category{特征线为指数曲线}
\begin{question}{特殊方程}
	
	\[
	\begin{cases}
		u_t + x \cdot Du = 0 & \mathbb{R}^n \times (0,+\infty) \\
		u = g & \mathbb{R}^n \times \{t=0\}
	\end{cases}
	\]
	此时系数非常向量，特征线满足 $\dfrac{dx(t)}{dt} = x(t)$.
\end{question}

\begin{minipage}[t]{0.7\textwidth}
	
	解特征方程：
	\[
	\frac{dx(t)}{dt} = x(t) \implies x(t) = x_0 e^t, \quad x_0 = \frac{x}{e^t} = x e^{-t}.
	\]
	沿特征线求导：
	\[
	\frac{d}{dt} u(x(t), t) = \frac{dx(t)}{dt} \cdot Du(x(t), t) + u_t(x(t), t) = x(t) \cdot Du(x(t),t) + u_t(x(t),t) = 0.
	\]
	$\Rightarrow u(x(t), t)$ 为常数.
	\[
	u(x,t) = u(x_0, 0) = g(x_0) = g\left(\frac{x}{e^t}\right) = g(x e^{-t}).
	\]
\end{minipage}

\subsection{非齐次问题}
\begin{knowledge}{非齐次传输方程}
	
	\[
	\begin{cases}
		u_t + \vec{b} \cdot Du = f & \mathbb{R}^n \times (0,+\infty) \\
		u = g & \mathbb{R}^n \times \{t=0\}
	\end{cases}
	\]
	源项 $f$ 表示外部输入. 仍沿特征线 $x(s) = \vec{b} s + x_0$ 求解.
\end{knowledge}
\begin{minipage}[t]{0.7\textwidth}
	
	沿特征线（以 $s$ 为参数）：
	\[
	\frac{d}{ds} u(x(s), s) = \vec{b} \cdot Du(x(s), s) + u_t(x(s), s) = f(x(s), s).
	\]
	两边对 $s$ 从 $0$ 到 $t$ 积分：
	\[
	u(x(t), t) - u(x_0, 0) = \int_0^t f(x(s), s)\, ds.
	\]
	由 $x = \vec{b} t + x_0$ 得 $x_0 = x - \vec{b} t$，$x(s) = \vec{b} s + x - \vec{b} t$，故
	\[
	u(x, t) = g(x - \vec{b} t) + \int_0^t f(\vec{b} s + x - \vec{b} t, s)\, ds.
	\]
	即解等于齐次行波解加上源项沿特征线的累积（Duhamel 原理）.
\end{minipage}

\section{第三讲\quad Laplace 方程与 Poisson 方程}
\subsection{方程与调和函数}
\begin{knowledge}{Laplace 方程与 Poisson 方程}
	
	Laplace 方程：
	\[
	\Delta u = 0, \quad u = u(x_1, \dots, x_n),
	\]
	其中 $\Delta u = \sum_{i=1}^n u_{x_i x_i}$ 为 Laplace 算子.
	
	Poisson 方程（非齐次）：
	\[
	-\Delta u = f.
	\]
\end{knowledge}
\begin{knowledge}{调和函数}
	
	开集 $U$ 上满足 $\Delta u = 0$ 的 $C^2$ 函数 $u$ 称为调和函数.
\end{knowledge}
\subsection{物理背景与 Gauss-Green 公式}
\begin{minipage}[t]{0.7\textwidth}
	
	\begin{knowledge}{物理意义}
		
		设 $u$ 表示某化学物质在 $x$ 处的浓度，$F$ 表示通量密度（向量值函数）. 由 Fick 定律：
		\[
		F = -c \cdot Du \quad (c > 0 \text{ 为扩散系数}).
		\]
		平衡状态时，任取闭包含于 $U$ 的区域 $V \subseteq U$，通过边界 $\partial V$ 的净流量为零（$\nu$ 为单位外法向）：
		\[
		\int_{\partial V} F \cdot \nu \, dS = 0.
		\]
	\end{knowledge}
	\begin{knowledge}{Gauss-Green 公式}
		
		设 $V$ 为具有光滑边界的有界区域，$u \in C^1(\bar{V})$，则
		\[
		\int_V u_{x_i} \, dx = \int_{\partial V} u \, \nu^i \, dS, \quad i = 1, \dots, n,
		\]
		其中 $\nu = (\nu^1, \dots, \nu^n)$ 为 $\partial V$ 的单位外法向.
	\end{knowledge}
\end{minipage}
\hfill
\begin{minipage}[t]{0.3\textwidth}
	
	\centering
	\begin{tikzpicture}[scale=0.9, baseline=(current bounding box.north)]
		\draw[thick, fill=LightBlue] (0,0) ellipse (1.5 and 1.1);
		\node at (0,0.15) {$V$};
		\node[below] at (0,-1.1) {$\partial V$};
		\coordinate (P) at (1.06,0.78);
		\fill (P) circle (1.5pt);
		\draw[->, thick, DarkRed] (P) -- (1.75,1.28) node[right] {$\nu$};
		\draw[->, thick, SkyBlue] (P) -- (1.6,0.15) node[right] {$F$};
	\end{tikzpicture}
\end{minipage}
\begin{question}{由平衡律导出 Laplace 方程}
	
	将 Gauss-Green 公式逐分量应用于向量场 $F$（即散度定理）：
	\[
	\int_{\partial V} F \cdot \nu \, dS = \sum_{i=1}^n \int_{\partial V} F_i \, \nu^i \, dS = \sum_{i=1}^n \int_V \frac{\partial F_i}{\partial x_i} \, dx = \int_V \operatorname{div} F \, dx.
	\]
	代入 $F = -c \, Du$：
	\[
	\int_{\partial V} F \cdot \nu \, dS = \sum_{i=1}^n \int_V -c \, u_{x_i x_i} \, dx = -c \int_V \Delta u \, dx.
	\]
	由平衡律，上式对任意 $V$ 均为 $0$；又 $\Delta u$ 连续，
	\[
	\therefore \Delta u = 0.
	\]
	即平衡态浓度必为调和函数.
\end{question}
\subsection{基本解：球对称约化}
\begin{minipage}[t]{0.7\textwidth}
	
	\begin{knowledge}{球对称约化}
		
		希望找到对称性好的解：只依赖于长度 $r = |x| = \sqrt{x_1^2 + \dots + x_n^2}$. 设
		\[
		u(x) = v(r), \quad v: \mathbb{R} \to \mathbb{R}, \ v \text{ 待定}.
		\]
	\end{knowledge}
	\begin{question}{计算 $\Delta u$}
		
		基本关系：
		\[
		\frac{\partial r}{\partial x_i} = \frac{1}{2}(x_1^2 + \dots + x_n^2)^{-\frac{1}{2}} \cdot 2x_i = \frac{x_i}{r}.
		\]
		$\therefore u_{x_i} = v'(r) \cdot \dfrac{x_i}{r}$.
		\[
		u_{x_i x_i} = v''(r) \cdot \frac{x_i^2}{r^2} + v'(r)\left(\frac{1}{r} - \frac{x_i^2}{r^3}\right).
		\]
		求和（注意 $\sum_{i=1}^n x_i^2 = r^2$）：
		\[
		\Delta u = \sum_{i=1}^n u_{x_i x_i} = v''(r) + v'(r)\left(\frac{n}{r} - \frac{1}{r}\right) = v''(r) + \frac{n-1}{r} v'(r) = 0.
		\]
	\end{question}
\end{minipage}
\hfill
\begin{minipage}[t]{0.3\textwidth}
	
	\centering
	\begin{tikzpicture}[scale=0.8, baseline=(current bounding box.north)]
		\draw[->, gray!40] (-2.2,0) -- (2.4,0) node[right] {$x_1$};
		\draw[->, gray!40] (0,-2.2) -- (0,2.4) node[above] {$x_2$};

		\draw[->, thick, SkyBlue] (0,0) -- (1.13,1.13);
		\fill (1.13,1.13) circle (2pt) node[above right] {$x$};
		\fill (0,0) circle (1.5pt) node[below left] {$O$};
		\node[SkyBlue] at (0.8,0.35) {$r=|x|$};
		\node[DarkRed, below] at (0,-1.9) {等值线（$n=2$ 示意）};
	\end{tikzpicture}
\end{minipage}
\begin{knowledge}{求解 ODE}
	
	\[
	\therefore \frac{v''(r)}{v'(r)} = \frac{1-n}{r}.
	\]
	\[
	\therefore (\log v'(r))' = \frac{1-n}{r}.
	\]
	积分：
	\[
	\log v'(r) = (1-n)\log r + C_1, \quad v'(r) = C_2 \cdot r^{1-n}.
	\]
	再积分：
	\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
		
		\item $n=2$：$v(r) = C_2 \cdot \ln r + C_3$.
		\item $n \geq 3$：$v(r) = C_2 \cdot \dfrac{1}{r^{n-2}} + C_3$.
	\end{enumerate}
\end{knowledge}
\begin{knowledge}{基本解}
	
	取适当常数，定义基本解：
	\[
	\Phi(x) =
	\begin{cases}
		-\dfrac{1}{2\pi} \ln|x|, & n=2, \\[8pt]
		\dfrac{1}{n(n-2)\alpha(n)} \cdot \dfrac{1}{|x|^{n-2}}, & n \geq 3,
	\end{cases}
	\]
	其中 $\alpha(n)$ 为 $\mathbb{R}^n$ 中单位球体积：
	\[
	\alpha(n) = \int_{|x|<1} dx = \frac{\pi^{\frac{n}{2}}}{\Gamma\left(\frac{n}{2}+1\right)}.
	\]
	注：$\Phi$ 在 $\mathbb{R}^n \setminus \{0\}$ 上调和，且在分布意义下满足 $-\Delta \Phi = \delta_0$，即原点处单位点源的响应.
\end{knowledge}
\subsection{Poisson 方程的解}
\begin{knowledge}{定理（Poisson 方程的解）}
	
	设 $f \in C_c^2(\mathbb{R}^n)$，即 $f$ 二次连续可微且具有紧支集（$\{f \neq 0\}$ 的闭包为紧集）. 定义
	\[
	u(x) = \int_{\mathbb{R}^n} \Phi(x-y) f(y)\, dy, \quad x \in \mathbb{R}^n.
	\]
	则 $u$ 是 Poisson 方程的解，即满足：
	\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
		
		\item $u \in C^2(\mathbb{R}^n)$；
		\item $-\Delta u = f$ 在 $\mathbb{R}^n$ 上成立.
	\end{enumerate}
\end{knowledge}


\subsection*{Poisson方程解的证明}
证明分三步.

\noindent\textbf{步1\quad 积分有意义（存在）.}

作换元 $y \to x-y$：
\[
\int_{\mathbb{R}^n} |\Phi(x-y) f(y)|\, dy = \int_{\mathbb{R}^n} |\Phi(y) f(x-y)|\, dy.
\]
$\because f$ 有紧支集，$\therefore \exists M>0$，使 $|y|>M$ 时 $f(x-y)=0$；又 $f$ 有界，$|f(x-y)| \leq C$.
\[
\therefore \int_{\mathbb{R}^n} |\Phi(y)|\, |f(x-y)|\, dy = \int_{|y|<M} |\Phi(y)|\, |f(x-y)|\, dy \leq C \int_{|y|<M} |\Phi(y)|\, dy.
\]
1$^\circ$ 当 $n=2$，取极坐标 $y_1=\rho\cos\theta$，$y_2=\rho\sin\theta$：
\[
\int_{|y|<M} |\ln|y||\, dy = \int_0^M d\rho \int_0^{2\pi} \rho\, |\ln \rho|\, d\theta = 2\pi \int_0^M \rho\, |\ln \rho|\, d\rho < \infty.
\]
2$^\circ$ 当 $n \geq 3$：
\[
\begin{aligned}
	\frac{1}{n(n-2)\alpha(n)} \int_{|y|<M} \frac{1}{|y|^{n-2}}\, dy &= \int_0^M d\rho \int_{\partial B(0,\rho)} \frac{1}{\rho^{n-2}}\, dS = \int_0^M \frac{1}{\rho^{n-2}} \cdot C_n \rho^{n-1}\, d\rho \\
	&= C_n \int_0^M \rho\, d\rho < \infty,
\end{aligned}
\]
其中 $C_n$ 为只与 $n$ 有关的常数（由 $\partial B(0,\rho)$ 面积 $= n\alpha(n)\rho^{n-1}$ 与系数 $\frac{1}{n(n-2)\alpha(n)}$ 合并而成）. $\therefore$ 积分绝对收敛，$u(x)$ 有意义.

\noindent\textbf{步2\quad $u$ 有偏导数 $u_{x_i}$，且 $u \in C^2(\mathbb{R}^n)$.}

取第 $i$ 个单位向量 $e_i$，考察差商：
\[
\frac{u(x+e_i h) - u(x)}{h} = \int_{\mathbb{R}^n} \Phi(y) \cdot \frac{f(x+e_i h - y) - f(x-y)}{h}\, dy.
\]
$\exists M>0$，使 $|y|>M$，$|h|<1$ 时有 $f(x+e_i h - y) = f(x-y) = 0$. 使用中值公式（$\theta \in (0,1)$）：
\[
\left| \frac{f(x+e_i h - y) - f(x-y)}{h} \right| = |f_{x_i}(x + \theta e_i h - y)| \leq \sup_{z \in \mathbb{R}^n} |f_{x_i}(z)|.
\]
\[
\therefore \left| \Phi(y) \cdot \frac{f(x+e_i h - y) - f(x-y)}{h} \right| \leq C \cdot |\Phi(y)| \cdot \mathbf{1}_{\{|y| \leq M\}},
\]
右端可积（步1）. $\therefore$ 由控制收敛定理：
\[
\lim_{h \to 0} \frac{u(x+e_i h) - u(x)}{h} = \int_{\mathbb{R}^n} \Phi(y) \lim_{h \to 0} \frac{f(x+e_i h - y) - f(x-y)}{h}\, dy = \int_{\mathbb{R}^n} \Phi(y) f_{x_i}(x-y)\, dy.
\]
同理：
\[
u_{x_i x_j}(x) = \int_{\mathbb{R}^n} \Phi(y) f_{x_i x_j}(x-y)\, dy,
\]
且 $u_{x_i x_j} \in C(\mathbb{R}^n)$（即 $\lim_{x \to x_0} u_{x_i x_j}(x) = u_{x_i x_j}(x_0)$），$\therefore u \in C^2(\mathbb{R}^n)$.

\noindent\textbf{步3\quad 证明 $-\Delta u = f$.}

先注意
\[
\frac{\partial f(x-y)}{\partial y_i} = f_{x_i}(x-y) \cdot (-1), \quad \frac{\partial^2 f(x-y)}{\partial y_i^2} = f_{x_i x_i}(x-y) \cdot (-1) \cdot (-1) = f_{x_i x_i}(x-y).
\]
\[
\therefore u_{x_i x_i} = \int_{\mathbb{R}^n} \Phi(y) \frac{\partial^2 f(x-y)}{\partial y_i^2}\, dy, \quad \Delta u = \int_{\mathbb{R}^n} \Phi(y) \Delta_y f(x-y)\, dy.
\]
把积分区域拆为 $B(0,\varepsilon)$ 与 $\mathbb{R}^n \setminus B(0,\varepsilon)$：
\[
\Delta u = \int_{B(0,\varepsilon)} \Phi(y) \Delta_y f(x-y)\, dy + \int_{\mathbb{R}^n \setminus B(0,\varepsilon)} \Phi(y) \Delta_y f(x-y)\, dy = I_\varepsilon + J_\varepsilon.
\]

\begin{minipage}[t]{0.7\textwidth}
	
	现证：当 $\varepsilon \to 0$ 时 $I_\varepsilon \to 0$，$J_\varepsilon \to -f(x)$.
	\[
	|I_\varepsilon| \leq \sum_{i=1}^n \sup_{z \in \mathbb{R}^n} |f_{x_i x_i}(z)| \cdot \int_{B(0,\varepsilon)} |\Phi(y)|\, dy \to 0.
	\]
	当 $n \geq 3$：
	\[
	\int_{B(0,\varepsilon)} \frac{1}{|y|^{n-2}}\, dy = \int_0^\varepsilon d\rho \int_{\partial B(0,\rho)} \frac{1}{\rho^{n-2}}\, dS = \int_0^\varepsilon \frac{1}{\rho^{n-2}} \cdot C_n \rho^{n-1}\, d\rho = \frac{C_n}{2} \varepsilon^2 \to 0
	\]
	（$n=2$ 时同理）.
	分部积分公式：
	\[
	\int_U u_{x_i} v\, dx = \int_{\partial U} u \cdot v \cdot \nu^i\, dS - \int_U u \cdot v_{x_i}\, dx.
	\]
	对 $J_\varepsilon$ 使用：
	\[
	\begin{aligned}
		J_\varepsilon &= \sum_{i=1}^n \int_{\mathbb{R}^n \setminus B(0,\varepsilon)} \Phi(y) \frac{\partial^2 f(x-y)}{\partial y_i^2}\, dy \\
		&= \sum_{i=1}^n \left[ \int_{\partial B(0,\varepsilon)} \Phi(y) \frac{\partial f(x-y)}{\partial y_i} \nu^i\, dS - \int_{\mathbb{R}^n \setminus B(0,\varepsilon)} \Phi_{y_i}(y) \frac{\partial f(x-y)}{\partial y_i}\, dy \right] \\
		&= L_\varepsilon + K_\varepsilon.
	\end{aligned}
	\]
	\[
	|L_\varepsilon| \leq \sum_{i=1}^n \sup_{z \in \mathbb{R}^n} |f_{x_i}(z)| \cdot \int_{\partial B(0,\varepsilon)} |\Phi(y)|\, dS.
	\]
	当 $n=2$ 时，$\int_{\partial B(0,\varepsilon)} |\ln|y||\, dS = |\ln \varepsilon| \cdot 2\pi \varepsilon \to 0$；
	当 $n \geq 3$ 时，$\int_{\partial B(0,\varepsilon)} \frac{1}{|y|^{n-2}}\, dS = \frac{1}{\varepsilon^{n-2}} \cdot C_n \varepsilon^{n-1} = C_n \varepsilon \to 0$.
	$\therefore L_\varepsilon \to 0$.
\end{minipage}

对 $K_\varepsilon$ 再次使用分部积分公式：
\[
K_\varepsilon = -\sum_{i=1}^n \left[ \int_{\partial B(0,\varepsilon)} \Phi_{y_i}(y) f(x-y) \nu^i\, dS - \int_{\mathbb{R}^n \setminus B(0,\varepsilon)} \Phi_{y_i y_i}(y) f(x-y)\, dy \right],
\]
其中第二项 $= \int \Delta \Phi \cdot f\, dy = 0$（$\Phi$ 在 $\mathbb{R}^n \setminus \{0\}$ 上调和）.
当 $n \geq 3$：
\[
\Phi_{y_i}(y) = \frac{\partial}{\partial y_i} \left[ \frac{1}{n(n-2)\alpha(n)} \cdot \frac{1}{|y|^{n-2}} \right] = -\frac{1}{n\alpha(n)} \cdot \frac{y_i}{|y|^n};
\]
在 $\partial B(0,\varepsilon)$ 上，区域 $\mathbb{R}^n \setminus B(0,\varepsilon)$ 的外法向指向原点：$\nu = -\frac{y}{|y|}$，$\nu^i = -\frac{y_i}{|y|}$.
\[
\begin{aligned}
	\therefore K_\varepsilon &= -\sum_{i=1}^n \int_{\partial B(0,\varepsilon)} \frac{1}{n\alpha(n)} \cdot \frac{y_i}{|y|^n} \cdot \frac{y_i}{|y|} \cdot f(x-y)\, dS_{(y)} = -\int_{\partial B(0,\varepsilon)} \frac{1}{n\alpha(n)} \cdot \frac{1}{|y|^{n-1}} f(x-y)\, dS_{(y)} \\
	&= -\frac{1}{n\alpha(n) \varepsilon^{n-1}} \int_{\partial B(0,\varepsilon)} f(x-y)\, dS_{(y)}.
\end{aligned}
\]
$\because \varepsilon^n \alpha(n) = \int_{|x|<\varepsilon} dx = \int_0^\varepsilon d\rho \int_{\partial B(0,\rho)} dS$，$\therefore n \varepsilon^{n-1} \alpha(n) = \int_{\partial B(0,\varepsilon)} dS$，即 $n\alpha(n)\varepsilon^{n-1}$ 是球面 $\partial B(0,\varepsilon)$ 的面积.
\[
\therefore |K_\varepsilon - (-f(x))| \leq \frac{1}{n\alpha(n)\varepsilon^{n-1}} \int_{\partial B(0,\varepsilon)} |f(x-y) - f(x)|\, dS_{(y)} \leq \sup_{y \in \partial B(0,\varepsilon)} |f(x-y) - f(x)| \to 0.
\]
当 $n=2$ 时，$\Phi(y) = -\frac{1}{2\pi} \ln|y|$，$\Phi_{y_i} = -\frac{1}{2\pi} \cdot \frac{y_i}{|y|^2}$，
\[
K_\varepsilon = -\sum_{i=1}^n \int_{\partial B(0,\varepsilon)} -\frac{1}{2\pi} \cdot \frac{y_i}{|y|^2} \cdot \left(-\frac{y_i}{|y|}\right) f(x-y)\, dS_{(y)} = -\frac{1}{2\pi} \cdot \frac{1}{\varepsilon} \int_{\partial B(0,\varepsilon)} f(x-y)\, dS_{(y)},
\]
（利用 $\sum_{i} y_i^2 = |y|^2 = \varepsilon^2$，且 $|\partial B(0,\varepsilon)| = 2\pi\varepsilon$）
\[
|K_\varepsilon - (-f(x))| \leq \sup_{y \in \partial B(0,\varepsilon)} |f(x-y) - f(x)| \to 0.
\]
综上，$\Delta u = I_\varepsilon + J_\varepsilon = I_\varepsilon + L_\varepsilon + K_\varepsilon \to -f(x)$（$\varepsilon \to 0$），而左端与 $\varepsilon$ 无关，
\[
\therefore -\Delta u = f. \quad \blacksquare
\]

\subsection{平均值公式}

\begin{minipage}[t]{0.7\textwidth}
	\begin{knowledge}
	\noindent\textbf{定理（平均值公式）}\quad 若 $u$ 为 $U$ 上的调和函数，则对每个 $B(x,r) \subseteq U$：
	\[
	u(x) = \frac{1}{n\alpha(n) r^{n-1}} \int_{\partial B(x,r)} u(y)\, dS_{(y)} = \frac{1}{\alpha(n) r^n} \int_{B(x,r)} u(y)\, dy,
	\]
	即调和函数在球心之值等于其球面平均，也等于其球体平均；反过来也成立（连续函数若对每个球都满足平均值性质，则必为调和函数）.
	\end{knowledge}
\end{minipage}

\noindent\textbf{证明}\quad 证明分五步：换元与面积元相消、积分号下求导、换回并用 Gauss-Green、定常数取极限、球体平均.

\noindent\textbf{步1（换元与面积元相消）}\quad 映射 $y = x + rz$ 把单位球面 $\partial B(0,1)$ 映到 $\partial B(x,r)$. 球面为 $(n-1)$ 维，每个切方向均被拉伸 $r$ 倍，故面积元满足
\[
dS_{(y)} = r^{n-1}\, dS_{(z)}.
\]
1代入球面平均的定义：
\[
\phi(r) = \frac{1}{n\alpha(n) r^{n-1}} \int_{\partial B(x,r)} u(y)\, dS_{(y)} = \frac{1}{n\alpha(n) r^{n-1}} \int_{\partial B(0,1)} u(x+rz)\, r^{n-1}\, dS_{(z)} = \frac{1}{n\alpha(n)} \int_{\partial B(0,1)} u(x+rz)\, dS_{(z)}.
\]
分母的 $r^{n-1}$ 与面积元带来的 $r^{n-1}$ 恰好相消；换元后积分域 $\partial B(0,1)$ 不再依赖 $r$.

\noindent\textbf{步2（积分号下求导）}\quad $\because u \in C^2(U)$，$\therefore$ 被积函数 $F(r,z) = u(x+rz)$ 与 $\dfrac{\partial F}{\partial r} = Du(x+rz) \cdot z$ 在 $(0,r_0) \times \partial B(0,1)$ 上连续，且积分域固定，由 Leibniz 法则可在积分号下求导：
\[
\phi'(r) = \frac{1}{n\alpha(n)} \int_{\partial B(0,1)} \frac{d}{dr} u(x+rz)\, dS_{(z)} = \frac{1}{n\alpha(n)} \int_{\partial B(0,1)} Du(x+rz) \cdot z\, dS_{(z)}.
\]

\noindent\textbf{步3（换回并用 Gauss-Green）}\quad 换回 $y = x+rz$，此时 $z = \dfrac{y-x}{r} = \nu$（$\partial B(x,r)$ 上的单位外法向），且 $dS_{(z)} = r^{-(n-1)} dS_{(y)}$：
\[
\phi'(r) = \frac{1}{n\alpha(n) r^{n-1}} \int_{\partial B(x,r)} Du(y) \cdot \frac{y-x}{r}\, dS_{(y)} = \frac{1}{n\alpha(n) r^{n-1}} \sum_{i=1}^n \int_{\partial B(x,r)} u_{y_i}(y)\, \nu^i\, dS_{(y)}.
\]
在 Gauss-Green 公式 $\displaystyle \int_U w_{x_i}\, dx = \int_{\partial U} w\, \nu^i\, dS$（即分部积分公式取 $v \equiv 1$，体积分项为 $0$）中取 $w = u_{y_i}$，$U = B(x,r)$：
\[
\int_{\partial B(x,r)} u_{y_i}\, \nu^i\, dS = \int_{B(x,r)} u_{y_i y_i}\, dy.
\]
对 $i$ 求和：
\[
\phi'(r) = \frac{1}{n\alpha(n) r^{n-1}} \sum_{i=1}^n \int_{B(x,r)} u_{y_i y_i}(y)\, dy = \frac{1}{n\alpha(n) r^{n-1}} \int_{B(x,r)} \Delta u\, dy = 0.
\]

\noindent\textbf{步4（定常数并取极限）}\quad $\therefore \phi'(r) \equiv 0$ 于 $(0,r_0)$，$\phi$ 为常数. 又 $\because u$ 连续，
\[
|\phi(r) - u(x)| \leq \sup_{y \in \partial B(x,r)} |u(y) - u(x)| \to 0 \quad (r \to 0),
\]
$\therefore \phi(r) = \lim_{r \to 0} \phi(r) = u(x)$，即球面平均 $= u(x)$.

\noindent\textbf{步5（球体平均）}\quad 由逐层分解（coarea）：
\[
\begin{aligned}
	\frac{1}{\alpha(n) r^n} \int_{B(x,r)} u(y)\, dy &= \frac{1}{\alpha(n) r^n} \int_0^r d\rho \int_{\partial B(x,\rho)} u(y)\, dS_{(y)} = \frac{1}{\alpha(n) r^n} \int_0^r u(x) \cdot n\alpha(n)\, \rho^{n-1}\, d\rho \\
	&= \frac{u(x)}{r^n} \int_0^r n \rho^{n-1}\, d\rho = u(x). \quad \blacksquare
\end{aligned}
\]

\noindent\textbf{接下来证明反过来也成立：连续函数若对每个球都满足平均值性质，则必为调和函数}

\begin{minipage}[t]{0.7\textwidth}
	
	\noindent\textbf{证明（反证法）}\quad 假设 $\Delta u(x) \neq 0$ 对某个 $x \in U$ 成立；不妨设 $\Delta u(x) > 0$（否则将 $u$ 换为 $-u$，论证完全相同）.
	
	$\because u \in C^2(U)$，$\therefore \Delta u$ 在 $U$ 上连续，$\therefore \exists r_0 > 0$，使 $\overline{B(x,r_0)} \subseteq U$，且
	\[
	\Delta u(y) > 0, \quad \forall y \in B(x, r_0).
	\]
	定义球面平均
	\[
	\phi(r) = \frac{1}{n\alpha(n) r^{n-1}} \int_{\partial B(x,r)} u(y)\, dS_{(y)}, \quad 0 < r \leq r_0.
	\]
	由假设 $\phi(r) = u(x)$ 与 $r$ 无关（为常数），$\therefore \phi'(r) \equiv 0$.
	
	另一方面，重复正方向步1--步3 的计算（换元、积分号下求导、对 $v \equiv 1$ 用 Gauss-Green 公式）：
	\[
	\phi'(r) = \frac{1}{n\alpha(n) r^{n-1}} \sum_{i=1}^n \int_{B(x,r)} u_{y_i y_i}(y)\, dy = \frac{1}{n\alpha(n) r^{n-1}} \int_{B(x,r)} \Delta u(y)\, dy.
	\]
	当 $0 < r < r_0$ 时，被积函数在 $B(x,r)$ 上恒正，$\therefore \phi'(r) > 0$.
	
	这与 $\phi'(r) = 0$ 矛盾（this is a contradiction）. $\therefore \Delta u(x) = 0$ 对一切 $x \in U$ 成立，即 $\Delta u = 0$，$u$ 为调和函数. $\blacksquare$

\noindent\textbf{注}\quad 综合正、反两个方向，对 $u \in C^2(U)$，下列三条等价：
\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
	
	\item $u$ 调和，即 $\Delta u = 0$；
	\item 对每个 $B(x,r) \subseteq U$，球面平均等于 $u(x)$；
	\item 对每个 $B(x,r) \subseteq U$，球体平均等于 $u(x)$.

\subsection{调和函数的性质}

\noindent\textbf{定理（强最大值原理）}\quad 设 $U$ 开、有界，$u \in C^2(U) \cap C(\bar{U})$ 是调和函数，则
\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
	
	\item $\max_{\bar{U}} u = \max_{\partial U} u$；
	\item 又若 $U$ 连通，且存在 $x_0 \in U$ 使 $u(x_0) = \max_{\bar{U}} u$（在内点处达到最大值），则 $u$ 在 $U$ 上为常数.
\end{enumerate}

（连通指：$U$ 不能写成两个非空不相交开集的并.）

\begin{minipage}[t]{0.7\textwidth}
	
	\noindent\textbf{证明 2$^\circ$}\quad $\because \bar{U}$ 紧、$u$ 连续，$\therefore$ 最大值取得到，设 $M = \max_{\bar{U}} u$，记最大值集 $S = \{x \in U : u(x) = M\} \neq \varnothing$.
	
	由 $u$ 的连续性，$S$ 在 $U$ 中为闭集；下证 $S$ 为开集：任取 $x \in S$，取 $B(x,r) \subseteq U$，由平均值公式（球体平均）：
	\[
	u(x) = \frac{1}{\alpha(n) r^n} \int_{B(x,r)} u(y)\, dy = M,
	\]
	即
	\[
	\frac{1}{\alpha(n) r^n} \int_{B(x,r)} \big(M - u(y)\big)\, dy = 0.
	\]
	$\because M - u(y) \geq 0$ 且连续，$\therefore u \equiv M$ 于 $B(x,r)$ 上，即 $B(x,r) \subseteq S$. $\therefore S$ 为开集.
	
	于是 $U = S \cup \{x \in U : u(x) < M\}$ 为两不相交开集之并；$\because U$ 连通且 $S \neq \varnothing$，$\therefore$ 后一集合必为空集（否则 $U$ 被写成两个非空不相交开集的并）. $\therefore u \equiv M$ 于 $U$ 上，即 $u$ 为常数.
\end{minipage}

		\node[gray] at (0,-2.05) {$S$ 在 $U$ 中既开又闭，故 $S=U$};
	\end{tikzpicture}
\end{minipage}

\noindent\textbf{证明 1$^\circ$}\quad 把 $U$ 写为互不相交连通分支之并：$U = \bigcup_i U_i$. 由 $\partial U \subseteq \bar{U}$，$\max_{\bar{U}} u \geq \max_{\partial U} u$ 显然；现证反向不等式.

设 $x_0 \in \bar{U}$ 使 $u(x_0) = \max_{\bar{U}} u$.
\begin{enumerate}[label=\arabic*$^\circ$, left=0pt]
	
	\item 若 $x_0 \in \partial U$，则 $\max_{\bar{U}} u = u(x_0) \leq \max_{\partial U} u$.
	\item 若 $x_0 \in U$，则 $x_0 \in U_i$（某个 $i$）；由 2$^\circ$，$u$ 在 $U_i$ 上为常数，故
	\[
	\max_{\bar{U}} u = u(x_0) = \max_{\partial U_i} u \leq \max_{\partial U} u
	\]
	（$\because \partial U_i \subseteq \partial U$）.
\end{enumerate}
综上两向不等式得 $\max_{\bar{U}} u = \max_{\partial U} u$. $\blacksquare$

\noindent\textbf{定理（严格正性）}\quad 设 $U$ 为有界连通开集，$u \in C^2(U) \cap C(\bar{U})$ 满足
\[
\begin{cases}
	\Delta u = 0 & \text{在 } U \text{ 中}, \\
	u = g & \text{在 } \partial U \text{ 上}.
\end{cases}
\]
若 $g \geq 0$ 且 $g$ 在 $\partial U$ 某处为正，则 $u$ 在 $U$ 内处处为正.

\noindent\textbf{证明}\quad 对 $-u$ 用强最大值原理：
\[
\min_{\bar{U}} u = \min_{\partial U} u = \min_{\partial U} g \geq 0,
\]
故 $u \geq 0$ 于 $\bar{U}$ 上. 若存在 $x_0 \in U$ 使 $u(x_0) = 0$，则最小值在内点达到；对 $-u$ 用 2$^\circ$ 得 $u \equiv 0$ 于 $U$ 上，再由连续性 $g = u|_{\partial U} \equiv 0$，与 $g$ 在某处为正矛盾. $\therefore u > 0$ 于 $U$ 内. $\blacksquare$

\noindent\textbf{定理（Poisson 方程的唯一性）}\quad 设 $U$ 有界开（不要求连通），$g \in C(\partial U)$，$f \in C(U)$，则至多存在一个 $u \in C^2(U) \cap C(\bar{U})$ 满足
\[
\begin{cases}
	-\Delta u = f & \text{in } U, \\
	u = g & \text{on } \partial U.
\end{cases}
\]

\noindent\textbf{证明}\quad 设 $u$ 与 $\hat{u}$ 都是解，令 $w = u - \hat{u}$，则
\[
\begin{cases}
	-\Delta w = 0 & \text{in } U, \\
	w = 0 & \text{on } \partial U.
\end{cases}
\]
由强最大值原理（分别用于 $w$ 与 $-w$）：
\[
\max_{\bar{U}} w = \max_{\partial U} w = 0, \quad \min_{\bar{U}} w = \min_{\partial U} w = 0,
\]
$\therefore w \equiv 0$，即 $u = \hat{u}$. $\blacksquare$

\noindent\textbf{定理（正则性）}\quad 设 $U$ 开，$u \in C(U)$ 在每个 $B(x,r) \subseteq U$ 上都满足平均值公式（特别地 $u$ 调和），则 $u \in C^\infty(U)$.

\begin{minipage}[t]{0.7\textwidth}
	
	\noindent\textbf{证明（Mollifier 磨光）}\quad 定义磨光子
	\[
	\eta(x) =
	\begin{cases}
		C_n \, e^{\frac{1}{|x|^2 - 1}}, & |x| < 1, \\
		0, & |x| \geq 1,
	\end{cases}
	\quad x \in \mathbb{R}^n,
	\]
	即取 $f(r) = C_n e^{\frac{1}{r-1}}$（$r<1$）、$0$（$r \geq 1$），$g(x) = |x|^2 = x_1^2 + \cdots + x_n^2$，则 $\eta(x) = f(g(x))$；由 $f$ 在 $r=1$ 处各阶导数均为 $0$ 知 $\eta \in C_c^\infty(\mathbb{R}^n)$，$\eta \geq 0$，$\mathrm{supp}\,\eta \subseteq \overline{B(0,1)}$. 常数 $C_n$ 由 $\int_{\mathbb{R}^n} \eta\, dx = 1$ 定出，由逐层分解（与 $\alpha(n)$ 的联系）：
	\[
	1 = \int_{\mathbb{R}^n} \eta(x)\, dx = C_n \int_{|x|<1} e^{\frac{1}{|x|^2-1}}\, dx = C_n \int_0^1 d\rho \int_{\partial B(0,\rho)} e^{\frac{1}{\rho^2-1}}\, dS = C_n \int_0^1 e^{\frac{1}{\rho^2-1}} \cdot n\alpha(n)\, \rho^{n-1}\, d\rho.
	\]
	对 $\varepsilon > 0$ 定义 $\eta_\varepsilon(x) = \dfrac{1}{\varepsilon^n} \eta\left(\dfrac{x}{\varepsilon}\right)$，$x \in \mathbb{R}^n$，则 $\eta_\varepsilon \in C_c^\infty(\mathbb{R}^n)$，$\displaystyle\int_{\mathbb{R}^n} \eta_\varepsilon(x)\, dx = 1$，$\mathrm{supp}\,\eta_\varepsilon \subseteq \overline{B(0,\varepsilon)}$.
	
	定义 $U_\varepsilon = \{x \in U : d(x, \partial U) > \varepsilon\}$，并在 $U_\varepsilon$ 上定义磨光
	\[
	u_\varepsilon(x) = (\eta_\varepsilon * u)(x) = \int_U \eta_\varepsilon(x-y) u(y)\, dy = \int_{B(x,\varepsilon)} \eta_\varepsilon(x-y) u(y)\, dy,
	\]
	由积分号下求导得 $u_\varepsilon \in C^\infty(U_\varepsilon)$.
\end{minipage}

\noindent\textbf{接证 $u_\varepsilon = u$ 于 $U_\varepsilon$ 上}\quad $\because \eta$ 为径向函数，可设 $\eta_\varepsilon(y) = \psi_\varepsilon(|y|)$；换元 $y \to x-y$ 并逐层分解：
\[
\begin{aligned}
	u_\varepsilon(x) &= \int_{B(0,\varepsilon)} \eta_\varepsilon(y)\, u(x-y)\, dy = \int_0^\varepsilon \psi_\varepsilon(\rho) \left( \int_{\partial B(x,\rho)} u(y)\, dS_{(y)} \right) d\rho \\
	&= \int_0^\varepsilon \psi_\varepsilon(\rho)\, n\alpha(n)\, \rho^{n-1}\, u(x)\, d\rho = u(x) \int_{\mathbb{R}^n} \eta_\varepsilon(y)\, dy = u(x),
\end{aligned}
\]
其中第二步用到球面平均值 $\dfrac{1}{n\alpha(n)\rho^{n-1}} \displaystyle\int_{\partial B(x,\rho)} u\, dS = u(x)$.

$\therefore u = u_\varepsilon \in C^\infty(U_\varepsilon)$；又 $\varepsilon$ 任意且 $U = \bigcup_{\varepsilon>0} U_\varepsilon$，$\therefore u \in C^\infty(U)$. $\blacksquare$
