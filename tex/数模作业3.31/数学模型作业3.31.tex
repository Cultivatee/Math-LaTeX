	\section{作业1：线性回归参数a和b的求解}
	
	\subsection{方法一：最小二乘法（定义式）}
	
	给定数据点 $(x_i, y_i), i=1,\dots,n$，拟合直线 $y = a + bx$，误差平方和为：
	\[
	Q = \sum_{i=1}^n (y_i - a - bx_i)^2
	\]
	
	由极值条件得到正规方程组：
	\[
	\begin{cases}
		na + b\sum_{i=1}^n x_i = \sum_{i=1}^n y_i \\[8pt]
		a\sum_{i=1}^n x_i + b\sum_{i=1}^n x_i^2 = \sum_{i=1}^n x_i y_i
	\end{cases}
	\]
	
	解为：
	\[
	b = \frac{l_{xy}}{l_{xx}},\quad a = \bar{y} - b\bar{x}
	\]
	
	其中：
	\[
	\bar{x} = \frac{1}{n}\sum_{i=1}^n x_i,\quad \bar{y} = \frac{1}{n}\sum_{i=1}^n y_i,\quad l_{xy} = \sum_{i=1}^n (x_i - \bar{x})(y_i - \bar{y}),\quad l_{xx} = \sum_{i=1}^n (x_i - \bar{x})^2
	\]
	
	\subsection{方法二：求偏导法}
	
	对 $Q(a,b)$ 求偏导并令其为零：
	\[
	\frac{\partial Q}{\partial a} = -2\sum_{i=1}^n (y_i - a - bx_i) = 0,\qquad 
	\frac{\partial Q}{\partial b} = -2\sum_{i=1}^n (y_i - a - bx_i)x_i = 0
	\]
	
	同样导出正规方程组：
	\[
	\begin{cases}
		na + b\sum_{i=1}^n x_i = \sum_{i=1}^n y_i \\[8pt]
		a\sum_{i=1}^n x_i + b\sum_{i=1}^n x_i^2 = \sum_{i=1}^n x_i y_i
	\end{cases}
	\]
	
	最终得到相同解：
	\[
	b = \frac{l_{xy}}{l_{xx}},\quad a = \bar{y} - b\bar{x}
	\]

	\subsection{模型拟合结果对比}
	
	采用三种数学模型对美国人口数据进行拟合分析：
	
	\subsubsection{线性模型}
	模型形式：$P(t) = a + b \cdot t$
	
	参数估计结果：
	\[
	\hat{b} \approx 1.4273,\qquad \hat{a} \approx -2605.3942
	\]
	
	回归方程：
	\[
	\hat{P}(t) = -2605.3942 + 1.4273\,t
	\]
	
	拟合优度：
	\[
	SSE_{\text{lin}} = \sum_{i=1}^n (P_i - \hat{P}_i)^2 \approx 20103.7404,\qquad R^2 = 1 - \frac{SSE}{SST} \approx 0.9210
	\]
	
	\subsubsection{指数模型}
	模型形式：$P(t) = A \cdot e^{B t}$
	
	通过对数变换 $\ln P = \ln A + B t$，使用最小二乘法得：
	\[
	B \approx 0.019105,\qquad A \approx e^{-32.4279} \approx 9.2064 \times 10^{-15}
	\]
	
	回归方程：
	\[
	\hat{P}(t) = 9.2064 \times 10^{-15} \cdot e^{0.019105 t}
	\]
	
	拟合优度：
	\[
	SSE_{\text{exp}} \approx 69280.1645,\qquad R^2 \approx 0.7276
	\]
	
	\subsubsection{Logistic阻滞增长模型}
	模型形式：\( P(t) = \dfrac{K}{1 + \left(\dfrac{K}{P_0} - 1\right)e^{-r(t-t_0)}} \)
	
	使用非线性最小二乘法估计参数（$t_0 = 1790$）：
	\[
	K \approx 497.6\ \text{（百万）},\qquad r \approx 0.020660,\qquad P_0 \approx 8.3\ \text{（百万）}
	\]
	
	回归方程：
	\[
	\hat{P}(t) = \frac{497.6}{1 + \left(\frac{497.6}{8.3} - 1\right)e^{-0.020660(t-1790)}}
	\]
	
	拟合优度：
	\[
	SSE_{\text{log}} \approx 514.2823,\qquad R^2 \approx 0.9980
	\]

	\subsection{模型拟合结果对比}
	
	采用三种数学模型对中国2000-2020年人口数据进行分析：
	
	\subsubsection{线性模型}
	模型形式：$P(t) = a + b \cdot t$
	
	参数估计结果：
	\[
	\hat{b} \approx 0.069582,\qquad \hat{a} \approx -126.4541
	\]
	
	回归方程：
	\[
	\hat{P}(t) = -126.4541 + 0.069582\,t
	\]
	
	拟合优度：
	\[
	SSE_{\text{lin}} = \sum_{i=1}^n (P_i - \hat{P}_i)^2 \approx 0.004412,\qquad R^2 = 1 - \frac{SSE}{SST} \approx 0.998818
	\]
	
	2020年预测值：$\hat{P}(2020) \approx 14.1014$ 亿，与实际值 14.1178 亿相比，绝对误差为 $-0.0164$ 亿，相对误差 $-0.1163\%$。
	
	\subsubsection{指数模型}
	模型形式：$P(t) = A \cdot e^{B t}$
	
	通过对数变换 $\ln P = \ln A + B t$，使用最小二乘法得：
	\[
	B \approx 0.005196,\qquad A \approx e^{-7.848} \approx 0.000390
	\]
	
	回归方程：
	\[
	\hat{P}(t) = 0.000390 \cdot e^{0.005196 t}
	\]
	
	拟合优度：
	\[
	SSE_{\text{exp}} \approx 0.006879,\qquad R^2 \approx 0.998157
	\]
	
	2020年预测值：$\hat{P}(2020) \approx 14.1136$ 亿，绝对误差 $-0.0042$ 亿，相对误差 $-0.0300\%$。
	
	\subsubsection{Logistic阻滞增长模型}
	模型形式：$P(t) = \dfrac{K}{1 + \left(\dfrac{K}{P_0} - 1\right)e^{-r(t-t_0)}}$
	
	使用非线性最小二乘法估计参数（$t_0 = 2000$）：
	\[
	K \approx 19.51\ \text{（亿）},\qquad r \approx 0.016623,\qquad P_0 \approx 12.6964\ \text{（亿）}
	\]
	
	回归方程：
	\[
	\hat{P}(t) = \frac{19.51}{1 + \left(\frac{19.51}{12.6964} - 1\right)e^{-0.016623(t-2000)}}
	\]
	
	拟合优度：
	\[
	SSE_{\text{log}} \approx 0.003433,\qquad R^2 \approx 0.999080
	\]
	
	2020年预测值：$\hat{P}(2020) \approx 14.0874$ 亿，绝对误差 $-0.0304$ 亿，相对误差 $-0.2157\%$。

	\subsection{结果分析}
	
	\begin{enumerate}
		\item \textbf{线性模型}：拟合效果良好（$R^2 = 0.998818$），2020年预测值略低于实际值。这说明在2000-2020年间，中国人口增长近似线性，年均增长约695.8万人。
		
		\item \textbf{指数模型}：拟合效果略低于线性模型（$R^2 = 0.998157$），2020年预测值相对误差最小（仅$-0.03\%$），但该模型假设恒定增长率，不适用于长期预测。
		
		\item \textbf{Logistic模型}：拟合效果最佳（$R^2 = 0.999080$），环境容纳量 $K \approx 19.51$ 亿，远高于当前人口，说明中国人口增长尚未出现明显的S形饱和特征。该模型符合人口增长规律，适用于中长期预测。
	\end{enumerate}
